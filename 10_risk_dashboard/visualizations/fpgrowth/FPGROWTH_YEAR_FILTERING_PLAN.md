# FP-Growth Year Filtering Implementation Plan

## Current State
- FP-Growth already processes data by `event_year` (2016, 2017, 2018, or "train" for combined)
- Visualizations generated by `py_helpers/create_fpgrowth_visualizations.py`
- **Current approach:** Separate PNG/HTML files per year (not combined into single interactive viz)
- **Files generated:** `{cohort}_{age_band}_{item_type}_combined_top_itemsets.png` (static)
- **Network HTML:** `{cohort}_{age_band}_{item_type}_target_rules_network.html` (already Plotly, but no year filter)

## Data Structure
FP-Growth itemsets stored by year:
```
10_risk_dashboard/visualizations/fpgrowth/outputs/
├── opioid_ed/
│   ├── target/
│   │   ├── 25_44/
│   │   │   ├── train/           # Combined 2016-2018
│   │   │   │   ├── drug_name_itemsets_lift_filtered.json
│   │   │   │   └── drug_name_rules_lift_filtered.json
│   │   │   ├── 2016/
│   │   │   ├── 2017/
│   │   │   └── 2018/
```

**Key:** `event_year` parameter determines which year's data is processed

## Challenge
- FP-Growth pipeline already separates by year during processing
- Visualizations are created per-year, not combined
- Need to: Load multiple years' itemsets, combine into single interactive viz

## Solution Approach

### Option 1: Multi-Year Data Loading in Visualization Script (RECOMMENDED)
Modify `create_fpgrowth_visualizations.py` to:
1. Load itemsets for all years (2016, 2017, 2018, train)
2. Combine into single dataset with `year` column
3. Create Plotly visualizations with year dropdown
4. Generate one interactive HTML per cohort/age_band (not per year)

### Option 2: Combine at Dashboard Request Time
- Lambda loads year-specific JSONs dynamically
- Builds combined visualization on-demand
- **Issue:** Slower, more complex Lambda logic

```python
def create_multi_year_fpgrowth_plots(
    base_dir: str,
    cohort_name: str,
    age_band: str,
    split_type: str = "target",
    item_types: Optional[List[str]] = None,
    output_dir: str = None,
    s3_upload: bool = True,
    s3_bucket: str = None,
    s3_prefix: str = None,
    top_n: int = 30,
) -> Dict[str, Any]:
    """
    Create year-filtered interactive FP-Growth visualizations.
    
    Loads itemsets from multiple event_years (2016, 2017, 2018, train),
    combines into single Plotly visualization with year dropdown filter.
    """
    years = ["train", 2016, 2017, 2018]
    year_labels = {
        "train": "All Years (2016-2018)",
        2016: "2016",
        2017: "2017",
        2018: "2018"
    }
    
    all_plots = {}
    
    for item_type in item_types or ["drug_name"]:
        # Load itemsets for each year
        itemsets_by_year = {}
        rules_by_year = {}
        
        for year in years:
            year_str = str(year) if isinstance(year, int) else year
            itemset_path = Path(base_dir) / cohort_name / split_type / age_band.replace("-", "_") / year_str / f"{item_type}_itemsets_lift_filtered.json"
            rules_path = Path(base_dir) / cohort_name / split_type / age_band.replace("-", "_") / year_str / f"{item_type}_rules_lift_filtered.json"
            
            if itemset_path.exists():
                with open(itemset_path) as f:
                    itemsets_by_year[year] = json.load(f)
            
            if rules_path.exists():
                with open(rules_path) as f:
                    rules_by_year[year] = json.load(f)
        
        # Create interactive itemsets bar chart with year dropdown
        itemsets_fig = _create_interactive_itemsets_plot(
            itemsets_by_year, cohort_name, age_band, item_type, top_n, year_labels
        )
        itemsets_html_path = Path(output_dir) / f"{cohort_name}_{age_band.replace('-', '_')}_{item_type}_itemsets_interactive.html"
        itemsets_fig.write_html(str(itemsets_html_path), include_plotlyjs='cdn')
        
        # Create interactive network with year dropdown
        network_fig = _create_interactive_network_plot(
            rules_by_year, cohort_name, age_band, item_type, year_labels
        )
        network_html_path = Path(output_dir) / f"{cohort_name}_{age_band.replace('-', '_')}_{item_type}_network_interactive.html"
        network_fig.write_html(str(network_html_path), include_plotlyjs='cdn')
        
        all_plots[item_type] = {
            "itemsets_interactive": str(itemsets_html_path),
            "network_interactive": str(network_html_path)
        }
    
    return all_plots


def _create_interactive_itemsets_plot(
    itemsets_by_year: Dict,
    cohort_name: str,
    age_band: str,
    item_type: str,
    top_n: int,
    year_labels: Dict
) -> go.Figure:
    """
    Create interactive Plotly bar chart of top itemsets with year filtering.
    """
    fig = go.Figure()
    
    years = list(itemsets_by_year.keys())
    
    for i, year in enumerate(years):
        itemsets = itemsets_by_year[year]
        
        # Get top N by support
        sorted_itemsets = sorted(itemsets, key=lambda x: x.get('support', 0), reverse=True)[:top_n]
        
        itemset_labels = [str(x.get('itemset', [])) for x in sorted_itemsets]
        support_values = [x.get('support', 0) for x in sorted_itemsets]
        
        fig.add_trace(go.Bar(
            y=itemset_labels,
            x=support_values,
            name=year_labels.get(year, str(year)),
            orientation='h',
            visible=(i == 0),  # Show "All Years" by default
            marker=dict(color='#3b82f6'),
            hovertemplate=(
                "<b>Itemset:</b> %{y}<br>"
                "<b>Support:</b> %{x:.4f}<br>"
                "<extra></extra>"
            )
        ))
    
    # Create year dropdown buttons
    updatemenus = [dict(
        active=0,
        buttons=[
            dict(
                label=year_labels.get(year, str(year)),
                method="update",
                args=[
                    {"visible": [j == i for j in range(len(years))]},
                    {"title": f"Top {top_n} Itemsets: {cohort_name} {age_band} - {year_labels.get(year, str(year))}"}
                ]
            )
            for i, year in enumerate(years)
        ],
        x=0.15,
        xanchor="left",
        y=1.15,
        yanchor="top"
    )]
    
    fig.update_layout(
        title=f"Top {top_n} Itemsets: {cohort_name} {age_band} - All Years (2016-2018)",
        xaxis_title="Support",
        yaxis_title="",
        updatemenus=updatemenus,
        height=800,
        margin=dict(l=300, r=50, t=100, b=50)
    )
    
    return fig


def _create_interactive_network_plot(
    rules_by_year: Dict,
    cohort_name: str,
    age_band: str,
    item_type: str,
    year_labels: Dict
) -> go.Figure:
    """
    Create interactive Plotly network graph with year filtering.
    
    Network shows association rules as directed graph:
    - Nodes: Items
    - Edges: Rules (antecedent → consequent)
    - Edge width: Confidence
    - Node size: Support
    """
    fig = go.Figure()
    
    years = list(rules_by_year.keys())
    
    for i, year in enumerate(years):
        rules = rules_by_year[year]
        
        # Build graph from rules (simplified - would need full networkx implementation)
        # For now, create placeholder trace structure
        
        # Extract nodes and edges
        nodes = set()
        edges = []
        for rule in rules[:50]:  # Limit to top 50 rules for readability
            antecedent = rule.get('antecedent', [])
            consequent = rule.get('consequent', [])
            confidence = rule.get('confidence', 0)
            
            nodes.update(antecedent + consequent)
            edges.append({
                'from': ','.join(antecedent),
                'to': ','.join(consequent),
                'confidence': confidence
            })
        
        # Create network layout (would use networkx in full implementation)
        # Placeholder: circular layout
        import math
        node_list = list(nodes)
        n = len(node_list)
        node_positions = {
            node: (math.cos(2 * math.pi * i / n), math.sin(2 * math.pi * i / n))
            for i, node in enumerate(node_list)
        }
        
        # Add edge traces
        edge_traces = []
        for edge in edges:
            from_pos = node_positions.get(edge['from'].split(',')[0], (0, 0))
            to_pos = node_positions.get(edge['to'].split(',')[0], (0, 0))
            
            edge_traces.append(go.Scatter(
                x=[from_pos[0], to_pos[0], None],
                y=[from_pos[1], to_pos[1], None],
                mode='lines',
                line=dict(width=edge['confidence'] * 10, color='#cccccc'),
                hoverinfo='none',
                showlegend=False,
                visible=(i == 0)
            ))
        
        fig.add_traces(edge_traces)
        
        # Add node trace
        node_x = [node_positions[node][0] for node in node_list]
        node_y = [node_positions[node][1] for node in node_list]
        
        fig.add_trace(go.Scatter(
            x=node_x,
            y=node_y,
            mode='markers+text',
            marker=dict(size=15, color='#3b82f6'),
            text=node_list,
            textposition='top center',
            visible=(i == 0),
            hovertemplate="<b>%{text}</b><extra></extra>"
        ))
    
    # Year dropdown
    updatemenus = [dict(
        active=0,
        buttons=[
            dict(
                label=year_labels.get(year, str(year)),
                method="update",
                args=[
                    {"visible": [j == i for j in range(len(years) * 51)]},  # 50 edges + 1 node trace per year
                    {"title": f"Association Rules Network: {cohort_name} {age_band} - {year_labels.get(year, str(year))}"}
                ]
            )
            for i, year in enumerate(years)
        ],
        x=0.15,
        y=1.05
    )]
    
    fig.update_layout(
        title=f"Association Rules Network: {cohort_name} {age_band} - All Years (2016-2018)",
        updatemenus=updatemenus,
        height=800,
        showlegend=False,
        xaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
        yaxis=dict(showgrid=False, zeroline=False, showticklabels=False)
    )
    
    return fig
```

```javascript
function renderFPGrowthVisualizations(data) {
  // Prioritize interactive HTML
  if (data.itemsets_interactive) {
    document.getElementById("fpgrowth-itemsets-image").innerHTML = 
      `<iframe src="${data.itemsets_interactive}" style="width: 100%; height: 900px; border: 1px solid #ddd;"></iframe>`;

    document.getElementById("fpgrowth-itemsets-image").innerHTML =

  // Network: prioritize interactive
  const networkEl = document.getElementById("fpgrowth-network-container");

      `<iframe src="${data.network_interactive}" style="width: 100%; height: 900px; border: 1px solid #ddd;"></iframe>`;

}
```
