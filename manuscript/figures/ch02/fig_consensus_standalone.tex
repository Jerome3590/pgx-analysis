% Standalone PDF for CH2 @fig-consensus — Model tournament → SHAP ∩ FFA → Causal Features.
% Build: pdflatex -interaction=nonstopmode -jobname=fig_consensus fig_consensus_standalone.tex
%
% Merges two TikZ designs: (A) hatched SHAP circle + solid FFA + clipped intersection (checkmark,
%   north-east line pattern, muted shapBlue/ffaPeach, L-shaped pointer to CAUSAL FEATURES);
%   (B) model pool → selection (Best → SHAP; fixed XGB → FFA) as ensemble-of-ensembles + FFA path.
% Graphic encodes: (1) tournament / best-of-three for robust SHAP; (2) fixed XGBoost for comparable FFA;
% (3) intersection as consensus safety gate (high SHAP ∧ population-consistent FFA; filters ``lucky'' correlations).
\documentclass[border=14pt]{standalone}
\usepackage{tikz}
\usepackage{amssymb}
\usetikzlibrary{patterns, calc, positioning, shapes.geometric, arrows.meta, shadows}

\definecolor{shapBlue}{HTML}{E8EDF2}
\definecolor{ffaPeach}{HTML}{EBCABF}
\definecolor{lineGray}{HTML}{333333}

\begin{document}
\begin{tikzpicture}[
  font=\sffamily\small,
  arr/.style={-{Stealth[length=2.2mm]}, thick},
]

    % ---------- Title: tournament-of-ensembles (DECIDE) ----------
    \node[font=\sffamily\bfseries, anchor=south, text=lineGray, align=center] (hdr) at (-0.2, 2.62) {%
        Tournament of Models (ensemble-of-ensembles)\\[0.12em]
        {\sffamily\bfseries\normalsize $\rightarrow$ SHAP $\cap$ FFA $\rightarrow$ causal consensus}};

    % ---------- 1. Model pool (inputs to tournament) ----------
    \node[draw=lineGray, fill=gray!5, inner sep=6pt, align=left, drop shadow={opacity=0.22, shadow xshift=0.6pt, shadow yshift=-0.8pt}] (models) at (-5.85, 0.15) {%
        \begin{tabular}{@{}l@{}}
        \textbf{Model pool} \\[0.15em]
        \hline \\[-0.5em]
        \footnotesize $\bullet$ CatBoost \\
        \footnotesize $\bullet$ XGBoost \\
        \footnotesize $\bullet$ XGBoost RF
        \end{tabular}};

    % ---------- 2. Selection: Best → SHAP; fixed XGB → FFA ----------
    \node[draw=lineGray, diamond, fill=blue!8, inner sep=1pt, minimum size=1.2cm, aspect=1.5, drop shadow={opacity=0.2, shadow xshift=0.5pt, shadow yshift=-0.7pt}] (best) at (-3.35, 0.9) {\textbf{\tiny Best}};
    \node[draw=lineGray, rectangle, fill=orange!10, minimum width=1.2cm, minimum height=0.58cm, drop shadow={opacity=0.2, shadow xshift=0.5pt, shadow yshift=-0.7pt}] (xgbffa) at (-3.35, -0.68) {\textbf{\small XGB}};

    \coordinate (fork) at ($(models.east)+(0.45,0)$);
    \draw[arr, lineGray] (models.east) -- (fork);
    \draw[arr, lineGray] (fork) |- (best.west);
    \draw[arr, lineGray] (fork) |- (xgbffa.west);

    \node[font=\tiny\bfseries, text=lineGray!88!black, anchor=south] at (-4.55, 1.42) {Selection};

    % ---------- 3. Venn filter: SHAP (hatched) from best ensemble; FFA (solid peach) from fixed XGB ----------
    \coordinate (Cshap) at (-0.05, 0);
    \coordinate (Cffa) at (1.55, 0);
    \def\R{1.2cm}

    \draw[fill=shapBlue, draw=lineGray, thick] (Cshap) circle (\R);
    \fill[pattern=north east lines, pattern color=lineGray!40] (Cshap) circle (\R);
    \node[font=\sffamily\bfseries] at ($(Cshap)+(-0.42,0)$) {\large SHAP};

    \draw[fill=ffaPeach, draw=lineGray, thick] (Cffa) circle (\R);
    \node[font=\sffamily\bfseries] at ($(Cffa)+(0.42,0)$) {\large FFA};

    \begin{scope}
        \clip (Cshap) circle (\R);
        \clip (Cffa) circle (\R);
        \fill[ffaPeach!80!black!20] (-2,-2) rectangle (3,2);
        \fill[pattern=north east lines, pattern color=lineGray!60] (-2,-2) rectangle (3,2);
        \node at ($(Cshap)!0.5!(Cffa)+(0,0.08)$) {\huge \checkmark};
    \end{scope}

    % ---------- 4. Connectors: tournament outputs into the filter ----------
    \draw[arr, blue!55!black] (best.east) to[out=0, in=135] ($(Cshap)+(135:\R)$);
    \draw[arr, orange!70!black] (xgbffa.east) to[out=0, in=225] ($(Cffa)+(225:\R)$);

    \node[below=0.12cm of best, font=\tiny, text=blue!65!black, align=center]
        {Source: Best Ensemble};
    \node[below=0.12cm of xgbffa, font=\tiny, text=orange!75!black, align=center]
        {Source: XGBoost Fixed (FFA)};

    % ---------- 5. L-shaped pointer + CAUSAL FEATURES (stacked label; geometry like centered Venn) ----------
    \coordinate (pointerBase) at ($(Cshap)!0.5!(Cffa)+(0,-0.4)$);
    \draw[thick, lineGray] (pointerBase) -- ++(0,-1.4) -- ++(0.2,0);

    \node[anchor=west, align=left, inner sep=2pt] at ($(pointerBase)+(0.2,-1.4)$) {%
        \fontsize{8}{9}\selectfont\bfseries CAUSAL\\[-0.2ex]
        \fontsize{8}{9}\selectfont\bfseries FEATURES};

    \fill[lineGray] ($(pointerBase)-(0.05,0.05)$) rectangle ($(pointerBase)+(0.05,0.05)$);

    % ---------- 6. Rationale strip (Robustness / Consistency / Consensus gate) ----------
    \node[
      anchor=north,
      text width=13.2cm,
      align=left,
      font=\sffamily\tiny,
      text=lineGray!92!black,
      yshift=-0.15cm,
    ] at ($(pointerBase)+(0,-2.12)$) {%
      \textbf{Robustness (SHAP):} Best-of-three (CatBoost / XGBoost / XGBoost RF) reduces model-specific bias; high SHAP under the selected ensemble strengthens a feature's claim as claims-level signal.\par\smallskip
      \textbf{Consistency (FFA):} FFA is locked to a standard XGBoost model so feature-frequency baselines stay comparable across temporal holdouts (e.g., 2018 vs.\ 2019).\par\smallskip
      \textbf{Consensus filter (safety gate):} In causal terms, the intersection is the gate: high SHAP with negligible FFA can reflect unstable, one-off patterns. Requiring both impact (SHAP) and repeated rule support (FFA) filters out ``lucky'' correlations in high-dimensional clinical data.};

\end{tikzpicture}
\end{document}
