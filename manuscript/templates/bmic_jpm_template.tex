%% Pandoc/Quarto LaTeX template for MDPI – Journal of Personalized Medicine (jpm)
%%
%% PREREQUISITES – obtain the MDPI LaTeX bundle (mdpi.cls + Definitions/ folder):
%%   Option A (Overleaf): https://www.overleaf.com/latex/templates/mdpi-article-template/fcpwsspfzsph
%%   Option B (MDPI site): https://www.mdpi.com/authors/latex  → download "LaTeX Template"
%%   Option C (GitHub):   https://github.com/metaphori/Template-LaTeX-MDPI
%%
%% Place mdpi.cls and the Definitions/ sub-folder next to this file OR install them
%% in your local TeX tree (e.g. ~/texmf/tex/latex/mdpi/).
%%
%% Build: quarto render <chapter>.qmd --to pdf   (uses xelatex by default)
%% ============================================================

%% NOTE: Definitions/mdpi.cls is resolved via TEXINPUTS – set before building:
%%   Linux/macOS : export TEXINPUTS=".:path/to/manuscript/templates//:$$TEXINPUTS"
%%   Windows PS  : $$env:TEXINPUTS = ".;C:\...\manuscript\templates\;;"
%%   Makefile    : see manuscript/Makefile (TEXINPUTS set automatically)
%%   build.ps1   : see manuscript/build.ps1 (TEXINPUTS set automatically)
%%
%% Definitions/ folder lives at: templates/Definitions/  (copied from phillipe/latex/)
\documentclass[%
  $for(classoption)$$classoption$,$endfor$%
  jpm,%          <- MDPI journal shortcode for Journal of Personalized Medicine
  $if(articletype)$$articletype$$else$article$endif$,%
  submit,%       <- becomes 'accept' when accepted by editorial office
  moreauthors%   <- use 'oneauthor' for single-author manuscript
]{Definitions/mdpi}

%% ---- Packages already loaded by mdpi.cls (do NOT reload) ----
%% fontenc, inputenc, calc, indentfirst, fancyhdr, graphicx, epstopdf,
%% lastpage, ifthen, lineno, float, amsmath, setspace, enumitem,
%% booktabs, titlesec, etoolbox, xcolor, multirow, microtype,
%% amsthm, natbib, hyperref, geometry, newfloat, caption

%% ---- Additional packages ----
\usepackage{amsfonts}
\usepackage{amssymb}
\usepackage{array}
\usepackage{longtable}
\usepackage{rotating}
\usepackage{tabularx}
\usepackage{lscape}

%% ---- Pandoc/Quarto compatibility shims (must come BEFORE header-includes) ----
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

%% mdpi.cls \DeclareFloatingEnvironment[]{listing} (via newfloat) pre-defines \listoflistings.
%% Quarto then fails with \newcommand*\listoflistings (already defined). Undefine it first.
\let\listoflistings\relax

$if(header-includes)$
$header-includes$
$endif$

$if(highlighting-macros)$
$highlighting-macros$
$endif$

%% ---- Article pagination metadata (updated by Editorial Office on acceptance) ----
\firstpage{1}
\makeatletter
\setcounter{page}{\@firstpage}
\makeatother
\pubvolume{$if(pubvolume)$$pubvolume$$else$1$endif$}
\issuenum{$if(issuenum)$$issuenum$$else$1$endif$}
\articlenumber{$if(articlenumber)$$articlenumber$$else$1$endif$}
\pubyear{$if(year)$$year$$else$2026$endif$}
\copyrightyear{$if(year)$$year$$else$2026$endif$}
%\externaleditor{Academic Editor: Firstname Lastname}
%% This cls uses \datereceived{}/\dateaccepted{}/\datepublished{} — \history is a no-op here
%% mdpi.cls line 649 omits \def\@datepublished{} — initialize it here to prevent maketitle crash
\makeatletter\def\@datepublished{}\makeatother
\providecommand{\history}[1]{}
%\updates{yes}  %% uncomment if a correction/update is available

%% ---- Title ----
\Title{$title$}

%% ---- Authors ----
$if(author)$
\Author{$for(author)$%
  $if(it.name)$$it.name$$else$$it$$endif$%
  $if(it.orcid)$\orcidA{}$endif$%
  $sep$, $endfor$}
\AuthorNames{$for(author)$%
  $if(it.name)$$it.name$$else$$it$$endif$%
  $sep$, $endfor$}
$else$
\Author{Author Name}
\AuthorNames{Author Name}
$endif$

%% ---- Affiliations ----
$if(affiliation)$
\address{%
$for(affiliation)$%
  \textsuperscript{$it.number$}\quad $it.name$%
  $if(it.email)$; \href{mailto:$it.email$}{$it.email$}$endif$\\%
$endfor$}
$else$
\address{%
  \textsuperscript{1}\quad Department, Institution, City, Country}
$endif$

%% ---- Corresponding author ----
$if(correspondence)$
\corres{Correspondence: \href{mailto:$correspondence$}{$correspondence$}}
$else$
\corres{Correspondence: author@institution.edu}
$endif$

%% ---- Abstract ----
$if(abstract)$
\abstract{$abstract$}
$endif$

%% ---- Keywords (semicolon-separated, 5–8 terms) ----
$if(keywords)$
\keyword{$for(keywords)$$keywords$$sep$; $endfor$}
$endif$

%% ============================================================
\begin{document}

$body$

%% ---- Abbreviations (appears in body, after main text) ----
$if(manuscript-abbreviations)$
\abbreviations{Abbreviations}{$manuscript-abbreviations$}
$endif$

%% \bibliography is required here — Quarto natbib does NOT auto-inject \bibdata into the .aux.
%% Paths are relative to the CH_X/ directory where bibtex runs.
\bibliography{../refs/discipline,../refs/bmic-jpm}

\end{document}
