%% Shared by cpt_psp_template.tex (CH2, CH4) and cts_template.tex (CH3).
%% Page-1 style aligned with MDPI first-page `plain` (see templates/Definitions/mdpi.cls):
%% foot rule, journal metadata left, DOI or page number right, minimal header.
%% Replaces default WileyNJDv5 \ps@titlepage (heavy footer + \vskip-40pt overlap risk).
%%
%% Wiley sets footskip≈48.5pt; fancy foot rule + footer line + long abbreviations
%% \footnotetext can exceed that band so two-column body text overprints the footer.
%% Increase the foot area and slightly tighten the type block so text stops above it.
\usepackage{fancyhdr}
\makeatletter
\@ifpackageloaded{geometry}{}{\usepackage{geometry}}
%% footskip: distance from last baseline (incl. footnotes) to footer baseline
\geometry{footskip=96pt}
\makeatother
\makeatletter
\fancypagestyle{titlepage}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0.4pt}%
  \fancyhead{\null\vspace{8pt}}%
  \fancyfoot[L]{%
    \fontsize{8}{8}\selectfont
    {\em\@journal\ }{\bfseries\@copyyear}, {\em\@volume}, %
    \ifx\@FirstPg\@empty 1\else\@FirstPg\fi
    \ifx\@LastPg\@empty\else--\@LastPg\fi
  }%
  \fancyfoot[R]{%
    \fontsize{8}{8}\selectfont
    \ifx\@doi\@empty
      \thepage
    \else
      \url{https://doi.org/\@doi}%
    \fi
  }%
}
\makeatother
