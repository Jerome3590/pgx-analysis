%% Loaded from cpt_psp_template.tex via \input (avoids Pandoc template parsing of \@).
%%
%% Wiley \maketitle must use \twocolumn[\@maketitle] so the title/abstract strip and
%% the two-column body share the correct page/column grid. Calling \@maketitle alone
%% breaks that grid and body text can sit on top of the title/abstract.
%%
%% The kernel's \@twocolumn begins with \clearpage, which ships an empty first page
%% when \maketitle is the first output. For this call only, \let\clearpage\relax
%% inside a group suppresses that spurious clear while keeping \twocolumn[\@maketitle].
\makeatletter
\renewcommand{\maketitle}{\par%
  \@afterindentfalse%
   \begingroup
    \renewcommand{\thefootnote}{\fnsymbol{footnote}}%
    \def\@makefnmark{\hbox to\z@{$\m@th^{\@thefnmark}$\hss}}%
    \long\def\@makefntext##1{\noindent
            \hbox to0.1em{\hss$\m@th^{\@thefnmark}$}##1}%
    \if@twocolumn
      \begingroup
      \let\clearpage\relax
      \twocolumn[\@maketitle]%
      \endgroup
    \else
      \newpage
      \global\@topnum\z@
      \@maketitle
    \fi
    \thispagestyle{titlepage}\@thanks%
    \insert\footins{\authfndefns}\par%
  \endgroup
  \setcounter{footnote}{0}%
  \let\thanks\relax
  \let\maketitle\relax\let\@maketitle\relax
  \gdef\@thanks{}\gdef\@author{}\gdef\@title{}%
  \@afterheading%
}
\makeatother
